// Проверка rm и chown — команды, изменяющие состояние VFS в памяти
ls

// удаление файла: до и после должно быть видно исчезновение из ls
rm a.txt
ls

// удаление второго файла
rm b.txt
ls

// смена владельца оставшегося файла
chown user1 notes.txt

// повторное удаление уже удалённого файла — ожидаемая ошибка
rm a.txt

// chown без указания файла — ожидаемая ошибка (не хватает аргументов)
chown user1

// chown несуществующего файла — ожидаемая ошибка
chown user1 nowhere.txt

// удаление вложенной директории (проверка rm для папок, если реализовано)
rm home

ls
exit
